DROP DATABASE IF EXISTS gestion_incidencias;
CREATE DATABASE gestion_incidencias
  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE gestion_incidencias;

-- ============ USUARIOS ============
CREATE TABLE usuarios (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(120) NOT NULL,
  email VARCHAR(150) NOT NULL UNIQUE,
  password_hash VARCHAR(255) NOT NULL,
  rol ENUM('admin','analista','viewer') DEFAULT 'analista',
  activo TINYINT DEFAULT 1,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ============ CATÁLOGOS ============
CREATE TABLE aplicativos (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL UNIQUE,
  activo TINYINT DEFAULT 1
);

CREATE TABLE categorias (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(80) NOT NULL UNIQUE,
  color VARCHAR(20) DEFAULT '#3b82f6'
);

CREATE TABLE sedes (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE analistas (
  id INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(120) NOT NULL,
  email VARCHAR(150) UNIQUE,
  nivel TINYINT NOT NULL,
  activo TINYINT DEFAULT 1
);

-- ============ TICKETS (ANTES de los INSERT) ============
CREATE TABLE tickets (
  id INT AUTO_INCREMENT PRIMARY KEY,
  codigo VARCHAR(20) NOT NULL UNIQUE,
  titulo VARCHAR(200) NOT NULL,
  aplicativo_id INT NOT NULL,
  categoria_id INT NOT NULL,
  sede_id INT NOT NULL,
  analista_id INT NULL,
  nivel TINYINT NOT NULL DEFAULT 1,
  prioridad ENUM('Baja','Media','Alta','Crítica') DEFAULT 'Media',
  estado ENUM('Abierto','En Proceso','Resuelto','Cerrado','Cancelado') DEFAULT 'Abierto',
  sla_horas INT NOT NULL DEFAULT 8,
  fecha_apertura DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  fecha_cierre DATETIME NULL,
  cumplio_sla TINYINT DEFAULT NULL,
  tiempo_atencion_horas DECIMAL(10,2) NULL,
  CONSTRAINT fk_ticket_app FOREIGN KEY (aplicativo_id) REFERENCES aplicativos(id),
  CONSTRAINT fk_ticket_cat FOREIGN KEY (categoria_id) REFERENCES categorias(id),
  CONSTRAINT fk_ticket_sede FOREIGN KEY (sede_id) REFERENCES sedes(id),
  CONSTRAINT fk_ticket_analista FOREIGN KEY (analista_id) REFERENCES analistas(id),
  INDEX idx_fecha (fecha_apertura),
  INDEX idx_estado (estado)
);

-- ============ SEED MASIVO: 1500 tickets ============
DELIMITER $$
CREATE PROCEDURE seed_tickets()
BEGIN
  DECLARE i INT DEFAULT 1;
  DECLARE v_apertura DATETIME;
  DECLARE v_cierre DATETIME;
  DECLARE v_estado VARCHAR(20);
  DECLARE v_cerrado TINYINT;
  DECLARE v_horas DECIMAL(10,2);

  WHILE i <= 1500 DO
    SET v_apertura = DATE_SUB(NOW(), INTERVAL FLOOR(RAND()*30) DAY);
    SET v_estado = ELT(FLOOR(1 + RAND()*5),
      'Resuelto','Cerrado','Cerrado','En Proceso','Abierto');
    SET v_cerrado = IF(v_estado IN ('Resuelto','Cerrado'), 1, 0);

    IF v_cerrado = 1 THEN
      SET v_horas = ROUND(0.5 + RAND()*20, 2);
      SET v_cierre = DATE_ADD(v_apertura, INTERVAL FLOOR(v_horas) HOUR);
    ELSE
      SET v_horas = NULL;
      SET v_cierre = NULL;
    END IF;

    INSERT INTO tickets (
      codigo, titulo, aplicativo_id, categoria_id, sede_id, analista_id,
      nivel, prioridad, estado, sla_horas, fecha_apertura, fecha_cierre,
      cumplio_sla, tiempo_atencion_horas
    ) VALUES (
      CONCAT('TK-', LPAD(i,5,'0')),
      CONCAT('Incidente de prueba #', i),
      FLOOR(1 + RAND()*10), FLOOR(1 + RAND()*6),
      FLOOR(1 + RAND()*6),  FLOOR(1 + RAND()*9),
      FLOOR(1 + RAND()*3),
      ELT(FLOOR(1 + RAND()*4),'Baja','Media','Alta','Crítica'),
      v_estado,
      ELT(FLOOR(1 + RAND()*3), 4, 8, 24),
      v_apertura, v_cierre,
      IF(v_cerrado = 1, IF(RAND() > 0.15, 1, 0), NULL),
      v_horas
    );
    SET i = i + 1;
  END WHILE;
END$$
DELIMITER ;

CALL seed_tickets();
DROP PROCEDURE seed_tickets;
